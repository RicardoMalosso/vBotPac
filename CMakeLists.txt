cmake_minimum_required(VERSION 3.24)
project(PInj LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "PInj requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
FetchContent_MakeAvailable(minhook)
add_library(pinj_bridge SHARED native/bridge.cpp)
target_link_libraries(pinj_bridge PRIVATE minhook bcrypt advapi32)
add_executable(pinj native/controller.cpp)
target_link_libraries(pinj PRIVATE bcrypt)
if(MSVC)
  target_compile_options(pinj PRIVATE /W4 /EHsc)
  target_compile_options(pinj_bridge PRIVATE /W4 /EHsc)
endif()


set(PINJ_PROFILE_HEADER "" CACHE FILEPATH "Generated executable compatibility header")
if(PINJ_PROFILE_HEADER)
  if(NOT EXISTS "${PINJ_PROFILE_HEADER}")
    message(FATAL_ERROR "Generated profile header does not exist")
  endif()
  file(TO_CMAKE_PATH "${PINJ_PROFILE_HEADER}" PINJ_PROFILE_HEADER_NORMALIZED)
  foreach(target pinj pinj_bridge)
    target_compile_definitions(${target} PRIVATE PINJ_PROFILE_HEADER="${PINJ_PROFILE_HEADER_NORMALIZED}")
  endforeach()
endif()
